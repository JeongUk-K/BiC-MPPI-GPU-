# -----------------------------------------------------------------------------
# Workspace-link-collision manipulator GPU examples
# -----------------------------------------------------------------------------
# Assumes your existing project already finds CUDA, Eigen3, OpenMP, curand, and
# cudart. In this repository, prefer:
#
#   bash build_gpu.sh manipulator_workspace_collision

set(MANIPULATOR_WORKSPACE_INCLUDE
    ${CMAKE_CURRENT_SOURCE_DIR}/src/manipulator/manipulator_workspace_collision_package/include
    ${CMAKE_CURRENT_SOURCE_DIR}/mppi/cuda-accel
    ${CMAKE_CURRENT_SOURCE_DIR}/include/fastsc
)
set(MANIPULATOR_WORKSPACE_SRC_DIR
    ${CMAKE_CURRENT_SOURCE_DIR}/src/manipulator/manipulator_workspace_collision_package/src
)
set(MANIPULATOR_WORKSPACE_EXAMPLE_DIR
    ${CMAKE_CURRENT_SOURCE_DIR}/src/manipulator/manipulator_workspace_collision_package/examples
)

add_executable(manipulator_mppi_workspace_example
    ${MANIPULATOR_WORKSPACE_EXAMPLE_DIR}/manipulator_mppi_workspace_example.cpp
    ${MANIPULATOR_WORKSPACE_SRC_DIR}/mppi_gpu.cu
)

add_executable(manipulator_logmppi_workspace_example
    ${MANIPULATOR_WORKSPACE_EXAMPLE_DIR}/manipulator_logmppi_workspace_example.cpp
    ${MANIPULATOR_WORKSPACE_SRC_DIR}/mppi_gpu.cu
    ${MANIPULATOR_WORKSPACE_SRC_DIR}/log_mppi_gpu.cu
)

add_executable(manipulator_clustermppi_workspace_example
    ${MANIPULATOR_WORKSPACE_EXAMPLE_DIR}/manipulator_clustermppi_workspace_example.cpp
    ${MANIPULATOR_WORKSPACE_SRC_DIR}/mppi_gpu.cu
    ${MANIPULATOR_WORKSPACE_SRC_DIR}/cluster_mppi_gpu.cu
)

add_executable(manipulator_bicmppi_workspace_example
    ${MANIPULATOR_WORKSPACE_EXAMPLE_DIR}/manipulator_bicmppi_workspace_example.cpp
    ${MANIPULATOR_WORKSPACE_SRC_DIR}/bi_mppi_gpu_legacy.cu
    ${MANIPULATOR_WORKSPACE_SRC_DIR}/bi_mppi_gpu.cu
)

add_executable(manipulator_mppi_pinkNplace_workspace_example
    ${MANIPULATOR_WORKSPACE_EXAMPLE_DIR}/pinkNplace/manipulator_mppi_pinkNplace_workspace_example.cpp
    ${MANIPULATOR_WORKSPACE_SRC_DIR}/mppi_gpu.cu
)

add_executable(manipulator_logmppi_pinkNplace_workspace_example
    ${MANIPULATOR_WORKSPACE_EXAMPLE_DIR}/pinkNplace/manipulator_logmppi_pinkNplace_workspace_example.cpp
    ${MANIPULATOR_WORKSPACE_SRC_DIR}/mppi_gpu.cu
    ${MANIPULATOR_WORKSPACE_SRC_DIR}/log_mppi_gpu.cu
)

add_executable(manipulator_clustermppi_pinkNplace_workspace_example
    ${MANIPULATOR_WORKSPACE_EXAMPLE_DIR}/pinkNplace/manipulator_clustermppi_pinkNplace_workspace_example.cpp
    ${MANIPULATOR_WORKSPACE_SRC_DIR}/mppi_gpu.cu
    ${MANIPULATOR_WORKSPACE_SRC_DIR}/cluster_mppi_gpu.cu
)

add_executable(manipulator_bicmppi_pinkNplace_workspace_example
    ${MANIPULATOR_WORKSPACE_EXAMPLE_DIR}/pinkNplace/manipulator_bicmppi_pinkNplace_workspace_example.cpp
    ${MANIPULATOR_WORKSPACE_SRC_DIR}/bi_mppi_gpu_legacy.cu
    ${MANIPULATOR_WORKSPACE_SRC_DIR}/bi_mppi_gpu.cu
)

foreach(target
        manipulator_mppi_workspace_example
        manipulator_logmppi_workspace_example
        manipulator_clustermppi_workspace_example
        manipulator_bicmppi_workspace_example
        manipulator_mppi_pinkNplace_workspace_example
        manipulator_logmppi_pinkNplace_workspace_example
        manipulator_clustermppi_pinkNplace_workspace_example
        manipulator_bicmppi_pinkNplace_workspace_example)
    target_include_directories(${target} PRIVATE ${MANIPULATOR_WORKSPACE_INCLUDE})
    target_link_libraries(${target} PRIVATE Eigen3::Eigen OpenMP::OpenMP_CXX curand cudart)
    set_target_properties(${target} PROPERTIES
        CUDA_SEPARABLE_COMPILATION ON
        CUDA_ARCHITECTURES native
    )
endforeach()
